<body>
  <h1>🎯 Restaurant Fortune Wheel</h1>
  <textarea id="restaurantList" placeholder="Enter restaurant names, one per line..."></textarea>
  <button onclick="setupWheel()">Load Restaurants</button>
  <button onclick="spinWheel()">Spin Wheel 🎡</button>

  <div class="wheel-container">
    <div class="pointer"></div>
    <canvas id="wheel" width="300" height="300"></canvas>
  </div>

  <div id="result"></div>

  <script>
    const canvas = document.getElementById('wheel');
    const ctx = canvas.getContext('2d');
    let restaurants = JSON.parse(localStorage.getItem('restaurants')) || ["Pizza Place", "Burger Barn", "Sushi Spot", "Taco Town"];
    let startAngle = 0, spinAngle = 0, spinTime = 0, spinTimeTotal = 0;

    function drawWheel() {
      const arc = Math.PI * 2 / restaurants.length;
      ctx.clearRect(0, 0, 300, 300);
      for (let i = 0; i < restaurants.length; i++) {
        const angle = startAngle + i * arc;
        ctx.beginPath();
        ctx.fillStyle = i % 2 === 0 ? "#ffb347" : "#ffcc33";
        ctx.moveTo(150, 150);
        ctx.arc(150, 150, 150, angle, angle + arc);
        ctx.fill();
        ctx.save();
        ctx.translate(150, 150);
        ctx.rotate(angle + arc / 2);
        ctx.textAlign = "right";
        ctx.fillStyle = "#222";
        ctx.font = "16px Segoe UI";
        ctx.fillText(restaurants[i], 140, 5);
        ctx.restore();
      }
    }

    function spinWheel() {
      if (restaurants.length === 0) return alert("Please add restaurants first!");
      spinAngle = Math.random() * 10 + 10;
      spinTime = 0;
      spinTimeTotal = Math.random() * 3000 + 4000;
      rotateWheel();
    }

    function rotateWheel() {
      spinTime += 30;
      if (spinTime >= spinTimeTotal) { stopRotateWheel(); return; }
      const spinEase = easeOut(spinTime, 0, spinAngle, spinTimeTotal);
      startAngle += (spinEase * Math.PI / 180);
      drawWheel();
      requestAnimationFrame(rotateWheel);
    }

    function stopRotateWheel() {
      const degrees = startAngle * 180 / Math.PI + 90;
      const arc = 360 / restaurants.length;
      const index = Math.floor((360 - (degrees % 360)) / arc) % restaurants.length;
      document.getElementById("result").textContent = `🎉 You got: ${restaurants[index]}!`;
    }

    function easeOut(t, b, c, d) {
      const ts = (t /= d) * t;
      const tc = ts * t;
      return b + c * (tc + -3 * ts + 3 * t);
    }

    function setupWheel() {
      const text = document.getElementById("restaurantList").value.trim();
      if (text) {
        restaurants = text.split("\n").map(r => r.trim()).filter(r => r);
        localStorage.setItem('restaurants', JSON.stringify(restaurants));
      }
      drawWheel();
      document.getElementById("result").textContent = "";
    }

    drawWheel();

    // Register service worker for offline support
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('service-worker.js');
    }
  </script>
